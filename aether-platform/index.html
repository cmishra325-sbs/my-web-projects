import { Aether } from "@aether/core";


const app = new Aether();


app.on("stripe.charge.successful", async (ctx) => {

await ctx.storeState("user_status", "upgrading");

await ctx.triggerPipeline("ai-onboard-sequence", { userId: ctx.payload.id });

});

// Orchestrate robust multi-step LLM chains with safety logic

const flow = app.pipeline("ai-onboard-sequence");


flow

.step("gpt-agent", async (node) => {

const prompt = "Analyze user company and craft email sequence";

return await node.llmCall("gpt-4o", prompt);

})
